AquaSentinel - all commands
(Windows PowerShell. On Mac/Linux use "source venv/bin/activate" instead of "venv\Scripts\activate".)

------------------------------------------------------------
1. ONE-TIME SETUP
------------------------------------------------------------
python -m venv venv
venv\Scripts\activate
pip install -r requirements-dev.txt

------------------------------------------------------------
2. DATA + FIRST MODEL
------------------------------------------------------------
dvc repro                                 # runs simulate -> train (only what changed)
# or step by step:
python -m src.data simulate               # winter + summer data, about 2 minutes
python -m src.train                       # trains, logs to MLflow, promotes v1

dvc dag                                   # show the pipeline
dvc metrics show                          # show the latest model metrics
dvc params diff                           # what changed in params.yaml since the last commit

------------------------------------------------------------
3. RUN THE WEBSITE + API
------------------------------------------------------------
python main.py                            # http://localhost:8000  (docs: /docs)

------------------------------------------------------------
4. MONITORING, RETRAINING, ROLLBACK
------------------------------------------------------------
python -m src.monitor                     # ingest new data -> drift check -> retrain if needed
python -m src.registry list               # model versions in the MLflow registry
python -m src.registry promote 2          # put version 2 into production
python -m src.registry rollback           # go back to the previous model

python -m ops.watchdog                    # auto-rollback (keep running next to the API)
python -m ops.watchdog --start-cmd "python main.py"   # also restarts the API if it crashes
python -m ops.log_anomaly --generate-demo # log anomaly chart -> reports/log_anomalies.png

------------------------------------------------------------
5. MLFLOW UI
------------------------------------------------------------
mlflow ui --backend-store-uri sqlite:///mlflow.db --port 5000    # http://localhost:5000

------------------------------------------------------------
6. TESTS + LINT
------------------------------------------------------------
pytest -q
flake8 .

------------------------------------------------------------
7. DOCKER (website + MLflow + Prometheus + Grafana)
------------------------------------------------------------
docker compose up -d --build
docker compose run --rm trainer           # data + first model inside the stack
docker compose down
# ready-made image from GitHub:
docker pull ghcr.io/shikhar0305/aquasentinel:latest

------------------------------------------------------------
8. AIRFLOW (optional)
------------------------------------------------------------
set AQUA_HOME=%cd%                        # Mac/Linux: export AQUA_HOME=$(pwd)
airflow standalone                        # set dags_folder to the dags/ folder first

------------------------------------------------------------
9. FEAST FEATURE STORE (optional)
------------------------------------------------------------
pip install -r requirements-feast.txt
python -m src.data features
cd feature_repo
feast apply
feast materialize-incremental 2030-01-01T00:00:00
cd ..
python feature_repo/demo_feast.py

------------------------------------------------------------
10. START AGAIN FROM v1 (before a demo)
------------------------------------------------------------
Remove-Item -Recurse -Force mlflow.db, mlruns, models, logs, reports, data\incoming, data\snapshots -ErrorAction SilentlyContinue
python -m src.train
